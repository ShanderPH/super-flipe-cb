cmake_minimum_required(VERSION 3.22)
project(superflip_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(superflip-native src/main.cpp)
target_link_libraries(superflip-native PRIVATE PkgConfig::CURL PkgConfig::JSON OpenSSL::Crypto)
target_compile_options(superflip-native PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(superflip-native PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(FONT REQUIRED IMPORTED_TARGET freetype2)
find_package(Threads REQUIRED)
target_link_libraries(superflip-native PRIVATE PkgConfig::SDL PkgConfig::FONT Threads::Threads)
add_library(sfcb-disc SHARED src/disc_vfs.cpp)
target_link_libraries(sfcb-disc PRIVATE PkgConfig::CURL PkgConfig::JSON OpenSSL::Crypto Threads::Threads)
set_target_properties(sfcb-disc PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
add_executable(native-probe src/probe.cpp)
target_link_libraries(native-probe PRIVATE PkgConfig::SDL PkgConfig::JSON PkgConfig::EGL PkgConfig::GLES ${CMAKE_DL_LIBS})
target_compile_options(native-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(native-probe PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS superflip-native native-probe RUNTIME DESTINATION .)
